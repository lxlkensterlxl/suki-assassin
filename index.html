      function bindEvents() {
        // Click Sukibot SVG to poke & slash
        dom.sukibotSvg.addEventListener("click", () => {
          triggerKatanaSlash();
          const pokeLine = decorateCuteGoth("Eek! Careful with my shadow katana, Senpai");
          appendMessage(1, "sukibot", pokeLine);
          speakCuteAnime(pokeLine);
        });

        dom.slashTriggerBtn.addEventListener("click", () => {
          triggerKatanaSlash();
          const line = decorateCuteGoth("Hyaaah! Crescent shadow strike");
          speakCuteAnime(line);
          showToast("Sukibot performed a blade slash!", "⚔️");
        });

        dom.voiceToggleBtn.addEventListener("click", () => {
          state.voiceEnabled = !state.voiceEnabled;
          if (!state.voiceEnabled && "speechSynthesis" in window) {
            window.speechSynthesis.cancel();
            stopLipSync();
          }
          saveState();
          renderTraitsAndStatus();
          showToast(state.voiceEnabled ? "Anime voice unmuted!" : "Anime voice muted.", state.voiceEnabled ? "🔊" : "🔇");
        });

        dom.pitchSlider.addEventListener("input", (e) => {
          state.voicePitch = parseFloat(e.target.value);
          dom.pitchValDisplay.textContent = state.voicePitch.toFixed(2);
          saveState();
        });

        dom.rateSlider.addEventListener("input", (e) => {
          state.voiceRate = parseFloat(e.target.value);
          dom.rateValDisplay.textContent = state.voiceRate.toFixed(2);
          saveState();
        });

        dom.testVoiceBtn.addEventListener("click", () => {
          if (!state.voiceEnabled) {
            state.voiceEnabled = true;
            renderTraitsAndStatus();
          }
          const sample = decorateCuteGoth("Nya~ Shadow blade ready! I am Sukibot");
          speakCuteAnime(sample);
        });

        // Chat Box 1 submission
        dom.chat1Form.addEventListener("submit", (e) => {
          e.preventDefault();
          const text = dom.chat1Input.value.trim();
          if (!text) return;
          dom.chat1Input.value = "";

          appendMessage(1, "user", text);
          const reply = generateGeneralReply(text);
          setTimeout(() => {
            appendMessage(1, "sukibot", reply);
            speakCuteAnime(reply);
          }, 180);
        });

        // Chat Box 2 submission (Self-Improvement AI Lab)
        dom.chat2Form.addEventListener("submit", (e) => {
          e.preventDefault();
          const text = dom.chat2Input.value.trim();
          if (!text) return;
          dom.chat2Input.value = "";

          appendMessage(2, "user", text);
          const reply = processSelfImprovementPrompt(text);
          setTimeout(() => {
            appendMessage(2, "sukibot", reply);
            speakCuteAnime(reply);
          }, 200);
        });

        // Quick self-upgrade prompt chips
        document.querySelectorAll(".quick-upgrade-chip").forEach((btn) => {
          btn.addEventListener("click", () => {
            const prompt = btn.getAttribute("data-upgrade-prompt") || "";
            if (!prompt) return;
            appendMessage(2, "user", prompt);
            const reply = processSelfImprovementPrompt(prompt);
            setTimeout(() => {
              appendMessage(2, "sukibot", reply);
              speakCuteAnime(reply);
            }, 180);
          });
        });

        // Clear chat buttons
        dom.clearChat1Btn.addEventListener("click", () => {
          state.chat1 = [{
            sender: "sukibot",
            text: decorateCuteGoth("Chat 1 cleared! What else shall we conquer"),
            time: nowTime()
          }];
          saveState();
          renderChats();
        });

        dom.clearChat2Btn.addEventListener("click", () => {
          state.chat2 = [{
            sender: "sukibot",
            text: "Self-Improvement Lab chat cleared! Ready for your next code upgrade instruction.",
            time: nowTime()
          }];
          saveState();
          renderChats();
        });

        // Inspector tabs
        dom.tabLogBtn.addEventListener("click", () => {
          dom.evolutionLogPanel.classList.remove("hidden");
          dom.liveCodePanel.classList.add("hidden");
          dom.tabLogBtn.className = "px-3 py-1.5 rounded-xl font-semibold bg-[#b026ff]/25 text-purple-100 border border-[#b026ff]/50 cursor-pointer";
          dom.tabCodeBtn.className = "px-3 py-1.5 rounded-xl font-semibold bg-crypt text-purple-300/75 hover:text-[#39ff14] border border-[#b026ff]/25 cursor-pointer";
        });

          dom.tabLogBtn.className = "px-3 py-1.5 rounded-xl font-semibold bg-crypt text-purple-300/75 hover:text-[#39ff14] border border-[#b026ff]/25 cursor-pointer";
          renderLiveCodeViewer();
        });

          if ("speechSynthesis" in window) window.speechSynthesis.cancel();

          state = deepClone(DEFAULT_STATE);
          saveState();

        });

        if ("speechSynthesis" in window) {
          selectBestAnimeVoice();
          window.speechSynthesis.onvoiceschanged = selectBestAnimeVoice;
